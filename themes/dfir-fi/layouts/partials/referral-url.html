{{- /* Returns the given URL with site.Params.referralQuery appended (e.g. utm_source=dfir.fi), so provider and training sites can see referrals. Uses "?" or "&" as needed and keeps any #fragment at the end. Usage: {{ partial "referral-url.html" .Params.website }} */ -}} {{- $url := . -}} {{- with site.Params.referralQuery -}} {{- $parts := split $url "#" -}} {{- $base := index $parts 0 -}} {{- $fragment := "" -}} {{- if gt (len $parts) 1 }}{{ $fragment = printf "#%s" (delimit (after 1 $parts) "#") }}{{ end -}} {{- $sep := cond (strings.Contains $base "?") "&" "?" -}} {{- $url = printf "%s%s%s%s" $base $sep . $fragment -}} {{- end -}} {{- return $url -}}
